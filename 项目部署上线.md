## docker部署步骤

### 需要准备的文件

#### 前端

- 打包好的dist包

- dockerfiles

```.dockerfiles
# ====== 运行阶段（本地已构建，仅需运行时环境） ======
# 前置条件：本地执行 npm run build 后，
#           将生成的 dist/ 文件夹复制到本文件同目录下

FROM nginx:1.24-alpine

# 复制本地预构建的前端产物
COPY dist/ /usr/share/nginx/html/

# 复制自定义 Nginx 配置
COPY nginx.conf /etc/nginx/conf.d/default.conf

# 暴露 80 端口
EXPOSE 80

# 启动 Nginx（前台运行）
CMD ["nginx", "-g", "daemon off;"]
```

- nginx配置文件

```.nginx
server {
    listen 80;
    server_name localhost;

    # 前端静态资源
    location / {
        root /usr/share/nginx/html;
        index index.html;
        try_files $uri $uri/ /index.html;
    }

    # 后端 API 反向代理（容器间通过 Docker 网络访问后端服务）
    location /api/ {
        proxy_pass http://yupao-backend:8080/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # 修正 Cookie Path，将后端的 /api 路径映射为根路径
        proxy_cookie_path /api /;

        proxy_connect_timeout 60s;
        proxy_read_timeout 120s;
        proxy_send_timeout 60s;
    }

    # 静态资源缓存
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        root /usr/share/nginx/html;
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # gzip 压缩
    gzip on;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml text/javascript image/svg+xml;
    gzip_min_length 1024;
}

```

#### 后端

- 需要在prod.yml中配置好需要在服务器使用的依赖

```.yml

# @from <a href="https://yupi.icu">编程导航知识星球</a>

spring:

  datasource:
    driver-class-name: com.mysql.cj.jdbc.Driver
    url: ${SPRING_DATASOURCE_URL:jdbc:mysql://localhost:3306/yupao?serverTimezone=Asia/Shanghai&useUnicode=true&characterEncoding=utf8&useSSL=false&allowPublicKeyRetrieval=true}
    username: ${SPRING_DATASOURCE_USERNAME:root}
    password: ${SPRING_DATASOURCE_PASSWORD:123456}

  # Session 配置（使用 Redis 存储 Session）
  session:
    store-type: redis
    timeout: 86400
  # Redis 配置 - 支持环境变量覆盖
  redis:
    host: ${SPRING_REDIS_HOST:localhost}
    port: ${SPRING_REDIS_PORT:6379}
    password: ${SPRING_REDIS_PASSWORD:}
    database: 0
server:
  address: 0.0.0.0
  servlet:
    context-path: /api
  session:
    cookie:
      domain:  #Cookie 的域名范围。空值表示 Cookie 只对当前访问的域名有效。 如果有了域名的话就写上域名，例：example.com
      有的时候需要频繁改动的话，在dockercompose中设置环境变量，就可以在服务器上直接更改，不用再打jar包再上传
      http-only: true
      secure: false   # 未配置 HTTPS 时必须为 false，配置 HTTPS 后改为 true
      same-site: lax  # secure=false 时不能用 none，否则浏览器拒绝 Cookie；HTTPS 后改为 none

```

- dockerfiles

```.dockerfiles
# ====== 运行阶段（本地已构建，仅需运行时环境） ======
# 前置条件：本地执行 mvn package -DskipTests 后，
#           将 target/yupao-backend-0.0.1-SNAPSHOT.jar 复制到本文件同目录下并重命名为 app.jar

FROM eclipse-temurin:8-jre-alpine

ENTRYPOINT ["sh", "-c", "java -Xms128m -Xmx256m -XX:+UseSerialGC -XX:+HeapDumpOnOutOfMemoryError -Djava.security.egd=file:/dev/./urandom -jar app.jar --spring.profiles.active=prod --server.port=${PORT:-8080}"]

```

- 为了方便部署，还得有dockercompose文件

```.dockercompose
version: '3.8'

services:
  # ========== MySQL 数据库 ==========
  mysql:
    image: mysql:8.0
    container_name: yupao-mysql
    restart: always
    environment:
      MYSQL_ROOT_PASSWORD: ${MYSQL_ROOT_PASSWORD:-yupao_root_2024}
      MYSQL_DATABASE: yupao
      MYSQL_USER: yupao
      MYSQL_PASSWORD: ${MYSQL_PASSWORD:-yupao_db_2024}
      TZ: Asia/Shanghai
    ports: -"127.0.0.1:3306:3306"  # 生产环境不暴露端口，容器间通过 yupao-network 内部通信
    volumes:
      - mysql_data:/var/lib/mysql
      - ./sql/create_table.sql:/docker-entrypoint-initdb.d/init.sql
    command: >
      --character-set-server=utf8mb4
      --collation-server=utf8mb4_unicode_ci
      --innodb-buffer-pool-size=128M
      --max-connections=50
      --table-open-cache=400
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      interval: 10s
      timeout: 5s
      retries: 10
    networks:
      - yupao-network

  # ========== Redis 缓存 ==========
  redis:
    image: redis:7-alpine
    container_name: yupao-redis
    restart: always
    command: redis-server --requirepass ${REDIS_PASSWORD:-yupao_redis_2024} --appendonly yes --maxmemory 128mb --maxmemory-policy allkeys-lru
    ports:
      # - "6379:6379"   # 生产环境不需要暴露到宿主机，容器内通过 yupao-network 访问即可
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "-a", "${REDIS_PASSWORD:-yupao_redis_2024}", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - yupao-network

  # ========== 后端服务 ==========
  # 部署前请确保 app.jar 已放置在当前目录下（由本地 mvn package 构建生成）
  backend:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: yupao-backend
    restart: always
    ports:
      - "8080:8080"
    environment:
      - SPRING_DATASOURCE_URL=jdbc:mysql://mysql:3306/yupao?serverTimezone=Asia/Shanghai&useUnicode=true&characterEncoding=utf8&useSSL=false&allowPublicKeyRetrieval=true
      - SPRING_DATASOURCE_USERNAME=yupao
      - SPRING_DATASOURCE_PASSWORD=${MYSQL_PASSWORD:-yupao_db_2024}
      - SPRING_REDIS_HOST=redis
      - SPRING_REDIS_PORT=6379
      - SPRING_REDIS_PASSWORD=${REDIS_PASSWORD:-yupao_redis_2024}
    depends_on:
      mysql:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - yupao-network

  # ========== 前端服务 ==========
  # 部署前请确保 dist/ 目录已放置在 ../yupao-frontend-master/ 下（由本地 npm run build 生成）
  frontend:
    build:
      context: ../yupao-frontend
      dockerfile: Dockerfile
    container_name: yupao-frontend
    restart: always
    ports:
      - "80:80"
    depends_on:
      - backend
    networks:
      - yupao-network

# ========== 数据卷 ==========
volumes:
  mysql_data:
  redis_data:

# ========== 网络 ==========
networks:
  yupao-network:
    driver: bridge

```

### 开始部署

1. 安装需要的工具

```.bash
# 1. 更新系统
sudo yum update -y

# 2. 安装必要工具
sudo yum install -y yum-utils curl git

# 3. 添加 Docker 阿里云镜像源（国内速度快）
sudo yum-config-manager --add-repo https://mirrors.aliyun.com/docker-ce/linux/centos/docker-ce.repo

# 4. 安装 Docker + Compose 插件
sudo yum install -y docker-ce docker-ce-cli containerd.io docker-compose-plugin

# 5. 配置 Docker 镜像加速 + 日志限制（2G 服务器必须）
sudo mkdir -p /etc/docker
sudo tee /etc/docker/daemon.json <<'EOF'
{
  "registry-mirrors": [
    "https://docker.1ms.run",
    "https://docker.xuanyuan.me"
  ],
  "log-driver": "json-file",
  "log-opts": {
    "max-size": "10m",
    "max-file": "3"
  }
}
EOF

# 6. 启动 Docker 并设置开机自启
sudo systemctl daemon-reload
sudo systemctl start docker
sudo systemctl enable docker

# 7. 验证安装
docker --version          # 应输出 Docker version 2x.x.x
docker compose version    # 应输出 Docker Compose version v2.x.x

# 8. 添加 2G Swap（2G 内存不够，必须加）
sudo fallocate -l 2G /swapfile
sudo chmod 600 /swapfile
sudo mkswap /swapfile
sudo swapon /swapfile
echo '/swapfile none swap sw 0 0' | sudo tee -a /etc/fstab

# 验证 swap
free -h
# 应看到 Swap: 2.0G
```

  2.本地上传文件

```.bash
# 上传命令示例
scp docker-compose.yml .env app.jar user@your-server:/home/user/deploy/
scp -r sql user@your-server:/home/user/deploy/

      ../yupao-frontend-master/nginx.conf \
      ../yupao-frontend-master/dist \
      user@your-server:/home/user/yupao-frontend/
```

  3.服务器上构建

```.bash
docker compose up -d --build
```





## docker常用命令：

```.bash
docker logs -f yupao-frontend(容器名) #查看容器日志

docker ps #查看正在运行的容器

docker ps -a #看所有的容器

docker compose build backend(compose.yml中已有的服务) #构建compose.yml中的服务

docker compose up -d backend  重新启动服务，-d表示在后台运行，日志服务不会占用端口



```
